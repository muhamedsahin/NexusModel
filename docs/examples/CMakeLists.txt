cmake_minimum_required(VERSION 3.20)
project(NexusDocsExamples LANGUAGES CXX)
set(NEXUS_MODEL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NEXUS_MODEL_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. nexusmodel)
add_executable(docs_quickstart quickstart.cpp)
target_link_libraries(docs_quickstart PRIVATE nexus_model::nexus_model)
add_executable(docs_train_mse train_mse.cpp)
target_link_libraries(docs_train_mse PRIVATE nexus_model::nexus_model)
enable_testing()
add_test(NAME quickstart COMMAND docs_quickstart)
add_test(NAME train_mse COMMAND docs_train_mse)
add_executable(docs_compare_cpu ../benchmarks/compare_cpu.cpp)
target_link_libraries(docs_compare_cpu PRIVATE nexus_model::nexus_model)
